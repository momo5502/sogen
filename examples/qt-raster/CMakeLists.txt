cmake_minimum_required(VERSION 3.26)
project(sogen_qt_raster_example LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Widgets)
qt_standard_project_setup()
qt_add_executable(calculator calculator.cpp)
target_link_libraries(calculator PRIVATE Qt6::Widgets)
qt_add_executable(window-behavior window_behavior.cpp)
target_link_libraries(window-behavior PRIVATE Qt6::Widgets)
if(WIN32)
  qt_add_plugin(qbridge PLUGIN_TYPE platforms CLASS_NAME BridgePlugin SHARED plugin.cpp)
  target_link_libraries(qbridge PRIVATE Qt6::GuiPrivate Qt6::CorePrivate)
  target_include_directories(qbridge PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../src/qt-raster-protocol")
  target_compile_definitions(qbridge PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  set_target_properties(qbridge PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/platforms" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/platforms")
endif()
